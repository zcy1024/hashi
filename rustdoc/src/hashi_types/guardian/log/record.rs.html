32/// Write: `LogMessage` -> `LogRecord` -> JSON body.
33/// Read: actual S3 key + JSON body -> untrusted `LogRecord` -> `VerifiedLogRecord`.
34#[derive(Debug)]
35pub struct LogRecord {
36    /// Final S3 destination selected before signing. Readers must compare this
37    /// signed intended key with the actual key returned by S3.
38    pub object_key: String,
39    pub session_id: SessionID,
40    pub timestamp_ms: UnixMillis,
41    pub message: VersionedLogMessage,
42    /// Present for signed logs; omitted for unsigned logs (currently only OIAttestationUnsigned).
43    pub signature: Option<GuardianSignature>,
44}
45
46#[derive(Serialize)]
47struct LogSigningPayload<'a> {
48    schema_version: u64,
49    session_id: &'a SessionID,
50    object_key: &'a str,
51    message: &'a VersionedLogMessage,
52}
53
54impl Serialize for LogRecord {
55    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
56    where
57        S: serde::Serializer,
58    {
59        #[derive(Serialize)]
60        struct LogRecordWire<'a, M> {
61            schema_version: u64,
62            object_key: &'a str,
63            session_id: &'a SessionID,
64            timestamp_ms: UnixMillis,
65            message: &'a M,
66            #[serde(with = "crate::guardian::serde_utils::option_guardian_signature")]
67            signature: &'a Option<GuardianSignature>,
68        }
69
70        match &self.message {
71            VersionedLogMessage::V1(message) => LogRecordWire {
72                schema_version: VersionedLogMessage::SCHEMA_VERSION_V1,
73                object_key: &self.object_key,
74                session_id: &self.session_id,
75                timestamp_ms: self.timestamp_ms,
76                message,
77                signature: &self.signature,
78            }
79            .serialize(serializer),
80            VersionedLogMessage::V2(message) => LogRecordWire {
81                schema_version: VersionedLogMessage::SCHEMA_VERSION_V2,
82                object_key: &self.object_key,
83                session_id: &self.session_id,
84                timestamp_ms: self.timestamp_ms,
85                message,
86                signature: &self.signature,
87            }
88            .serialize(serializer),
89        }
90    }
91}

93impl<'de> Deserialize<'de> for LogRecord {
94    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
95    where
96        D: serde::Deserializer<'de>,
97    {
98        #[derive(Deserialize)]
99        struct LogRecordWire {
100            schema_version: u64,
101            object_key: String,
102            session_id: SessionID,
103            timestamp_ms: UnixMillis,
104            message: Value,
105            #[serde(with = "crate::guardian::serde_utils::option_guardian_signature")]
106            signature: Option<GuardianSignature>,
107        }
108
109        let raw = LogRecordWire::deserialize(deserializer)?;
110        let message = match raw.schema_version {
111            VersionedLogMessage::SCHEMA_VERSION_V1 => {
112                serde_json::from_value::<LogMessageV1>(raw.message)
113                    .map(VersionedLogMessage::V1)
114                    .map_err(D::Error::custom)?
115            }
116            VersionedLogMessage::SCHEMA_VERSION_V2 => {
117                serde_json::from_value::<LogMessage>(raw.message)
118                    .map(VersionedLogMessage::V2)
119                    .map_err(D::Error::custom)?
120            }
121            version => {
122                return Err(D::Error::custom(format!(
123                    "unsupported log schema version: {version}"
124                )));
125            }
126        };
127
128        Ok(Self {
129            object_key: raw.object_key,
130            session_id: raw.session_id,
131            timestamp_ms: raw.timestamp_ms,
132            message,
133            signature: raw.signature,
134        })
135    }
136}

142/// A log record whose message signature and writing session's attestation/PCRs
143/// have both been verified. Its message is normalized to the current schema.
144/// If a future schema cannot be converted losslessly, this type should retain
145/// `VersionedLogMessage` and leave version acceptance to callers instead.
146#[derive(Debug)]
147pub struct VerifiedLogRecord {
148    pub object_key: String,
149    pub session_id: SessionID,
150    pub timestamp_ms: UnixMillis,
151    pub message: LogMessage,
152    pub build_pcrs: BuildPcrs,
153}

155impl VerifiedLogRecord {
156    pub fn new(
157        object_key: String,
158        session_id: SessionID,
159        timestamp_ms: UnixMillis,
160        message: LogMessage,
161        build_pcrs: BuildPcrs,
162    ) -> Self {
163        Self {
164            object_key,
165            session_id,
166            timestamp_ms,
167            message,
168            build_pcrs,
169        }
170    }
171}

173impl LogRecord {
174    pub fn new(
175        session_id: SessionID,
176        message: LogMessage,
177        signing_key: &GuardianSignKeyPair,
178    ) -> Self {
179        Self::new_at_timestamp(session_id, message, signing_key, now_timestamp_ms())
180    }
181
182    pub fn new_at_timestamp(
183        session_id: SessionID,
184        message: LogMessage,
185        signing_key: &GuardianSignKeyPair,
186        timestamp_ms: UnixMillis,
187    ) -> Self {
188        let message = VersionedLogMessage::V2(message);
189        let object_key = message
190            .object_key_pattern(&session_id, timestamp_ms)
191            .finalize();
192        if message.is_allowed_unsigned() {
193            Self::unsigned(session_id, message, timestamp_ms, object_key)
194        } else {
195            Self::signed(session_id, message, signing_key, timestamp_ms, object_key)
196        }
197    }
198
199    pub fn object_key(&self) -> &str {
200        &self.object_key
201    }
202
203    pub fn object_lock_duration(&self, policy: S3ObjectLockPolicy) -> Duration {
204        policy.duration_for(self.message.log_type())
205    }
206
207    fn signed(
208        session_id: SessionID,
209        message: VersionedLogMessage,
210        signing_key: &GuardianSignKeyPair,
211        timestamp_ms: UnixMillis,
212        object_key: String,
213    ) -> Self {
214        let mut record = Self {
215            object_key,
216            session_id,
217            timestamp_ms,
218            message,
219            signature: None,
220        };
221        record.signature = Some(sign_intent(
222            &record.signing_payload(),
223            timestamp_ms,
224            signing_key,
225        ));
226        record
227    }
228
229    fn unsigned(
230        session_id: SessionID,
231        message: VersionedLogMessage,
232        timestamp_ms: UnixMillis,
233        object_key: String,
234    ) -> Self {
235        assert!(
236            message.is_allowed_unsigned(),
237            "message must be Init(OIAttestationUnsigned)"
238        );
239        Self {
240            object_key,
241            session_id,
242            timestamp_ms,
243            message,
244            signature: None,
245        }
246    }
247
248    pub fn verify(
249        self,
250        pub_key: &GuardianPubKey,
251    ) -> GuardianResult<(SessionID, UnixMillis, LogMessage)> {
252        if self.message.is_allowed_unsigned() {
253            return Err(InvalidS3Log(
254                "expected signed log record but message is unsigned".into(),
255            ));
256        }
257        self.validate_object_key()?;
258        self.validate_session_id(pub_key)?;
259        let timestamp_ms = self.timestamp_ms;
260        let signature = self
261            .signature
262            .as_ref()
263            .ok_or_else(|| InvalidS3Log("missing log signature".into()))?;
264        verify_intent(&self.signing_payload(), timestamp_ms, signature, pub_key)
265            .map_err(|e| InvalidS3Log(format!("invalid log signature: {e}")))?;
266
267        let message = self
268            .message
269            .into_current()
270            .map_err(|e| InvalidS3Log(format!("log schema conversion failed: {e}")))?;
271        Ok((self.session_id, timestamp_ms, message))
272    }
273
274    /// Validates the unsigned OI-attestation record's envelope and canonical
275    /// session. The Nitro attestation itself must be authenticated separately.
276    pub fn validate_unsigned(self) -> GuardianResult<(SessionID, UnixMillis, LogMessage)> {
277        if !self.message.is_allowed_unsigned() {
278            return Err(InvalidS3Log(
279                "expected unsigned log record but message requires a signature".into(),
280            ));
281        }
282        if self.signature.is_some() {
283            return Err(InvalidS3Log(
284                "unsigned log record must not contain a signature".into(),
285            ));
286        }
287        self.validate_object_key()?;
288        let init = match &self.message {
289            VersionedLogMessage::V1(LogMessageV1::Init(init)) => init,
290            VersionedLogMessage::V2(LogMessage::Init(init)) => init,
291            _ => unreachable!("is_allowed_unsigned only permits an init message"),
292        };
293        let super::messages::InitLogMessage::OIAttestationUnsigned {
294            signing_public_key, ..
295        } = init.as_ref()
296        else {
297            unreachable!("is_allowed_unsigned only permits OIAttestationUnsigned");
298        };
299        self.validate_session_id(signing_public_key)?;
300        let message = self
301            .message
302            .into_current()
303            .map_err(|e| InvalidS3Log(format!("log schema conversion failed: {e}")))?;
304        Ok((self.session_id, self.timestamp_ms, message))
305    }
306
307    /// Rejects a record whose signed intended key differs from the key at which
308    /// the S3 reader found it.
309    pub fn validate_actual_object_key(&self, actual_object_key: &str) -> GuardianResult<()> {
310        if self.object_key != actual_object_key {
311            return Err(InvalidS3Log(format!(
312                "S3 object key mismatch: record contains {}, actual key is {actual_object_key}",
313                self.object_key
314            )));
315        }
316        Ok(())
317    }
318
319    fn validate_object_key(&self) -> GuardianResult<()> {
320        match self
321            .message
322            .object_key_pattern(&self.session_id, self.timestamp_ms)
323        {
324            ObjectKeyPattern::Fixed(expected) if self.object_key != expected => {
325                return Err(InvalidS3Log(format!(
326                    "non-canonical S3 object key: got {}, expected {expected}",
327                    self.object_key
328                )));
329            }
330            ObjectKeyPattern::RandomSuffix(prefix) if !self.object_key.starts_with(&prefix) => {
331                return Err(InvalidS3Log(format!(
332                    "non-canonical S3 object key: got {}, expected prefix {prefix}",
333                    self.object_key
334                )));
335            }
336            _ => {}
337        }
338        Ok(())
339    }
340
341    fn validate_session_id(&self, signing_public_key: &GuardianPubKey) -> GuardianResult<()> {
342        let canonical_session_id = SessionID::from_signing_pubkey(signing_public_key);
343        if self.session_id != canonical_session_id {
344            return Err(InvalidS3Log(format!(
345                "session ID mismatch: record contains {}, signing public key derives {canonical_session_id}",
346                self.session_id
347            )));
348        }
349        Ok(())
350    }
351
352    fn signing_payload(&self) -> LogSigningPayload<'_> {
353        LogSigningPayload {
354            schema_version: self.message.schema_version(),
355            session_id: &self.session_id,
356            object_key: &self.object_key,
357            message: &self.message,
358        }
359    }
