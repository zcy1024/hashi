165    /// Invoke a peer's bridge-service signing RPC, retrying on transient
166    /// transport failures. `call` is handed a freshly-fetched client each
167    /// attempt so the retry reconnects (tonic reconnects lazily) rather than
168    /// reusing the connection the peer just tore down. Returns `None` once
169    /// attempts are exhausted or on a non-transport error.
170    async fn call_peer_with_retry<Resp, F, Fut>(
171        inner: &Arc<Hashi>,
172        validator: Address,
173        what: &str,
174        mut call: F,
175    ) -> Option<tonic::Response<Resp>>
176    where
177        F: FnMut(BridgeServiceClient<BoxedChannel>) -> Fut,
178        Fut: Future<Output = Result<tonic::Response<Resp>, tonic::Status>>,
179    {
180        const MAX_ATTEMPTS: u32 = 3;
181        for attempt in 1..=MAX_ATTEMPTS {
182            let Some(client) = inner.onchain_state().bridge_service_client(&validator) else {
183                error!("Cannot find bridge-service client for validator: {validator:?}");
184                return None;
185            };
186            match call(client).await {
187                Ok(response) => return Some(response),
188                Err(status) if attempt < MAX_ATTEMPTS && is_retriable_transport(&status) => {
189                    warn!(
190                        "Failed to get {what} from {validator} (attempt {attempt}/{MAX_ATTEMPTS}): \
191                         {status}; retrying on a fresh connection"
192                    );
193                    tokio::time::sleep(Duration::from_millis(50 * u64::from(attempt))).await;
194                }
195                Err(status) => {
196                    error!("Failed to get {what} from {validator}: {status}");
197                    return None;
198                }
199            }
200        }
201        None
202    }

212    #[tracing::instrument(name = "leader", skip_all)]
213    async fn run(mut self) {
214        info!("Starting leader service");
215
216        // Wait for DKG to complete before processing any checkpoints.
217        let mpc_handle = self.inner.mpc_handle().expect("MpcHandle not initialized");
218        info!("Waiting for MPC key to become available...");
219        mpc_handle.wait_for_key_ready().await;
220        info!("MPC key is ready, starting leader loop");
221
222        let mut checkpoint_rx = self.inner.onchain_state().subscribe_checkpoint();
223        let mut btc_block_rx = self.inner.btc_monitor().subscribe_block_height();
224        let mut was_leader = false;
225
226        loop {
227            trace!("Waiting for next checkpoint or task completion...");
228            tokio::select! {
229                wait_result = checkpoint_rx.changed() => {
230                    if let Err(e) = wait_result {
231                        error!("Error waiting for checkpoint change: {e}");
232                        break;
233                    }
234                    let (checkpoint_height, checkpoint_timestamp_ms) = {
235                        let checkpoint_info = checkpoint_rx.borrow_and_update();
236                        (checkpoint_info.height, checkpoint_info.timestamp_ms)
237                    };
238
239                    let is_leader = self.is_current_leader(checkpoint_height);
240                    self.inner.metrics.is_leader.set(i64::from(is_leader));
241                    if is_leader {
242                        was_leader = true;
243                        debug!("Checkpoint {checkpoint_height}: We are the leader node");
244                    } else {
245                        if was_leader {
246                            self.reset_approved_deposit_metrics();
247                            was_leader = false;
248                        }
249                        trace!("We are not the leader node");
250                        continue;
251                    }
252
253                    self.check_reconcile_guardian_committee();
254                    self.process_unapproved_withdrawal_requests(checkpoint_timestamp_ms);
255                    self.process_approved_withdrawal_requests(checkpoint_timestamp_ms);
256                    self.process_unsigned_withdrawal_txns();
257                    self.process_signed_withdrawal_txns();
258                    self.check_delete_expired_deposit_requests(checkpoint_timestamp_ms);
259                    self.check_delete_proposals(checkpoint_timestamp_ms);
260                    self.check_cleanup_spent_utxos(checkpoint_timestamp_ms);
261                    self.process_stale_unapproved_deposits_if_new_epoch();
262                    self.process_approved_deposit_requests();
263                }
264                wait_result = btc_block_rx.changed() => {
265                    if let Err(e) = wait_result {
266                        error!("Error waiting for Bitcoin block height change: {e}");
267                        break;
268                    }
269
270                    let block_height = *btc_block_rx.borrow_and_update();
271                    let checkpoint_height = checkpoint_rx.borrow().height;
272
273                    self.schedule_withdrawal_checks_for_btc_block();
274
275                    if self.is_current_leader(checkpoint_height) {
276                        debug!("New Bitcoin block {block_height}: processing deposit requests");
277                        self.process_deposits_on_bitcoin_block();
278                    }
279                }
280                Some(result) = self.unapproved_deposit_tasks.join_next() => {
281                    self.handle_completed_unapproved_deposit_task(result);
282                    while let Some(result) = self.unapproved_deposit_tasks.try_join_next() {
283                        self.handle_completed_unapproved_deposit_task(result);
284                    }
285                }
286                Some(result) = self.approved_deposit_tasks.join_next() => {
287                    self.handle_completed_approved_deposit_task(result);
288                    while let Some(result) = self.approved_deposit_tasks.try_join_next() {
289                        self.handle_completed_approved_deposit_task(result);
290                    }
291                }
292                Some(result) = self.withdrawal_signing_tasks.join_next() => {
293                    self.handle_completed_withdrawal_signing_task(result);
294                }
295                Some(result) = self.withdrawal_broadcast_tasks.join_next() => {
296                    self.handle_completed_withdrawal_broadcast_task(result);
297                }
298                Some(result) = self.withdrawal_btc_block_check_tasks.join_next() => {
299                    self.handle_completed_withdrawal_btc_block_check_task(result);
300                }
301                Some(result) = OptionFuture::from(self.withdrawal_approval_task.as_mut()) => {
302                    self.withdrawal_approval_task = None;
303                    Self::log_task_result("withdrawal_approval", result);
304                }
305                Some(result) = OptionFuture::from(self.withdrawal_commitment_task.as_mut()) => {
306                    self.withdrawal_commitment_task = None;
307                    Self::log_task_result("withdrawal_commitment", result);
308                }
309                Some(result) = OptionFuture::from(self.deposit_gc_task.as_mut()) => {
310                    self.deposit_gc_task = None;
311                    Self::log_task_result("deposit_gc", result);
312                }
313                Some(result) = OptionFuture::from(self.proposal_gc_task.as_mut()) => {
314                    self.proposal_gc_task = None;
315                    Self::log_task_result("proposal_gc", result);
316                }
317                Some(result) = OptionFuture::from(self.utxo_cleanup_gc_task.as_mut()) => {
318                    self.utxo_cleanup_gc_task = None;
319                    // Re-arm the scan when the task did work (more may exist
320                    // past the per-GC cap, or new spends raced in) or failed
321                    // (retry after backoff); a barren scan stays disarmed
322                    // until a withdrawal confirms. Rescheduling is left to
323                    // the leader-gated checkpoint arm: respawning here ran
324                    // ungated (leader or not) with no backoff, which kept a
325                    // node resubmitting cleanups indefinitely.
326                    match &result {
327                        Ok(Ok(0)) => self.utxo_cleanup_retry.clear(),
328                        Ok(Ok(_)) => {
329                            self.utxo_cleanup_retry.clear();
330                            self.utxo_cleanup_scan_needed = true;
331                        }
332                        _ => {
333                            self.utxo_cleanup_retry.record_failure(
334                                garbage_collection::UtxoCleanupErrorKind::Failed,
335                                checkpoint_rx.borrow().timestamp_ms,
336                            );
337                            self.utxo_cleanup_scan_needed = true;
338                        }
339                    }
340                    Self::log_task_result("utxo_cleanup_gc", result);
341                }
342                Some(result) = OptionFuture::from(self.guardian_committee_reconcile_task.as_mut()) => {
343                    self.guardian_committee_reconcile_task = None;
344                    // On failure, clear the epoch gate so the next tick retries
345                    // (e.g. transient guardian downtime); success holds the gate
346                    // until the hashi epoch advances again.
347                    if !matches!(&result, Ok(Ok(()))) {
348                        self.last_guardian_reconcile_epoch = None;
349                    }
350                    Self::log_task_result("guardian_committee_reconcile", result);
351                }
352
353            }
354        }
355    }

357    fn log_task_result<T>(label: &str, result: Result<anyhow::Result<T>, tokio::task::JoinError>) {
358        match result {
359            Ok(Ok(_)) => {}
360            Ok(Err(err)) => error!("{label} task failed: {err:#?}"),
361            Err(err) if err.is_panic() => error!("{label} task panicked: {err}"),
362            Err(err) => error!("{label} task failed to join: {err}"),
363        }
364    }

380    /// Whether this node is the leader for the given checkpoint. An
381    /// associated function (rather than a method) so long-running leader
382    /// tasks can re-check leadership mid-flight and stop driving work that
383    /// has rotated to another node.
384    pub(super) fn node_is_leader(inner: &Arc<Hashi>, checkpoint_height: u64) -> bool {
385        if inner.onchain_state().state().hashi().config.paused() {
386            debug!("Bridge is paused, not acting as leader");
387            return false;
388        }
389
390        match inner.config.force_run_as_leader() {
391            ForceRunAsLeader::Always => return true,
392            ForceRunAsLeader::Never => return false,
393            ForceRunAsLeader::Default => (),
394        }
395
396        let Some(committee) = inner.onchain_state().current_committee() else {
397            // TODO: do we need to do anything when bootstrapping? At genesis there is no committee.
398            return false;
399        };
400        let this_validator_address = inner
401            .config
402            .validator_address()
403            .expect("No configured validator address");
404        let Some(this_validator_idx) = committee
405            .index_of(&this_validator_address)
406            .map(|i| i as u64)
407        else {
408            // We are not in the committee yet, so we cannot be the leader
409            return false;
410        };
411        let num_validators = committee.members().len() as u64;
412
413        let current_turn = checkpoint_height / NUM_CONSECUTIVE_LEADER_CHECKPOINTS;
414        let is_leader = (current_turn % num_validators) == this_validator_idx;
415
416        trace!("Node index {this_validator_idx} is leader node: {is_leader}");
417        is_leader
418    }

421fn parse_member_signature(
422    member_signature: hashi_types::proto::MemberSignature,
423) -> anyhow::Result<MemberSignature> {
424    let epoch = member_signature
425        .epoch
426        .ok_or(anyhow::anyhow!("No epoch in MemberSignature"))?;
427    let address_string = member_signature
428        .address
429        .ok_or(anyhow::anyhow!("No address in MemberSignature"))?;
430    let address = address_string
431        .parse::<Address>()
432        .map_err(|e| anyhow::anyhow!("Unable to parse Address: {}", e))?;
433    let signature = BLS12381Signature::from_bytes(
434        member_signature
435            .signature
436            .ok_or(anyhow::anyhow!("No signature in MemberSignature"))?
437            .as_bytes(),
438    )?;
439    Ok(MemberSignature::new(epoch, address, signature))
440}

442/// Whether a failed peer RPC is worth retrying. Under sustained load a peer's
443/// HTTP/2 server tears the whole multiplexed connection down — `GoAway`
444/// (surfaced as `Internal`), broken pipe (`Unknown`), or the usual
445/// `Unavailable` — failing every in-flight request to that peer at once. The
446/// peer signing RPCs are idempotent, so retrying these transport-class codes is
447/// safe and lets the request land on a fresh connection.
448fn is_retriable_transport(status: &tonic::Status) -> bool {
449    matches!(
450        status.code(),
451        tonic::Code::Unavailable
452            | tonic::Code::Unknown
453            | tonic::Code::Internal
454            | tonic::Code::Cancelled
455            | tonic::Code::DeadlineExceeded
456    )
457}

459#[cfg(test)]
460mod tests {
461    use super::is_retriable_transport;
462    use tonic::Code;
463    use tonic::Status;
464
465    #[test]
466    fn classifies_transport_errors_as_retriable() {
467        // A peer tearing the connection down surfaces as these codes: GoAway ->
468        // Internal, broken pipe -> Unknown, plus the usual transient codes.
469        for code in [
470            Code::Unavailable,
471            Code::Unknown,
472            Code::Internal,
473            Code::Cancelled,
474            Code::DeadlineExceeded,
475        ] {
476            assert!(
477                is_retriable_transport(&Status::new(code, "boom")),
478                "{code:?} should be retried"
479            );
480        }
481        // Genuine application rejections from the peer must not be retried.
482        for code in [
483            Code::InvalidArgument,
484            Code::FailedPrecondition,
485            Code::PermissionDenied,
486            Code::NotFound,
487            Code::AlreadyExists,
488        ] {
489            assert!(
490                !is_retriable_transport(&Status::new(code, "nope")),
491                "{code:?} should not be retried"
492            );
493        }
494    }
